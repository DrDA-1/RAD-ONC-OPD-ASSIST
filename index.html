        function calculate() {

            // Read raw parameters

            const isFemale = document.querySelector('input[name="sex"]:checked').value === 'female';

            const age = parseFloat(document.getElementById('age').value);

            let rawScr = parseFloat(document.getElementById('creatinine').value);

            let rawHeight = parseFloat(document.getElementById('height').value);

            let rawWeight = parseFloat(document.getElementById('weight').value);

 

            if (isNaN(age) || isNaN(rawScr) || isNaN(rawHeight) || isNaN(rawWeight) || rawScr <= 0 || age <= 0 || rawHeight <= 0 || rawWeight <= 0) {

                return;

            }

 

            // Standardize metric base units (mg/dL, cm, kg)

            const scr_mgdl = state.scrUnit === 'umol' ? rawScr / 88.4 : rawScr;

            const height_cm = state.heightUnit === 'in' ? rawHeight * 2.54 : rawHeight;

            const height_in = height_cm / 2.54;

            const weight_kg = state.weightUnit === 'lbs' ? rawWeight / 2.20462 : rawWeight;

 

            // 1. Quetelet Index (BMI)

            const height_m = height_cm / 100;

            const bmi = weight_kg / (height_m * height_m);

 

            // 2. Devine Ideal Body Weight (IBW)

            const heightOver60 = Math.max(0, height_in - 60);

            let ibw = isFemale ? 45.5 + (2.3 * heightOver60) : 50.0 + (2.3 * heightOver60);

            if (ibw < 10) ibw = weight_kg;

 

            // 3. Adjusted Body Weight (ABW-40)

            const abw = ibw + 0.4 * (weight_kg - ibw);

 

            // 4. BSA Calculations

            const bsa_mosteller = Math.sqrt((height_cm * weight_kg) / 3600);

            const bsa_dubois = 0.007184 * Math.pow(weight_kg, 0.425) * Math.pow(height_cm, 0.725);

 

            // 5. Cockcroft-Gault Clearances

            const sexFactorCG = isFemale ? 0.85 : 1.0;

            const crcl_actual = ((140 - age) * weight_kg) / (72 * scr_mgdl) * sexFactorCG;

           

            const effective_ibw = weight_kg < ibw ? weight_kg : ibw;

            const effective_abw = weight_kg < ibw ? weight_kg : abw;

 

            const crcl_ibw = ((140 - age) * effective_ibw) / (72 * scr_mgdl) * sexFactorCG;

            const crcl_adj = ((140 - age) * effective_abw) / (72 * scr_mgdl) * sexFactorCG;

 

            let recNote = "";

            if (weight_kg < ibw) {

                recNote = "Actual weight < IBW: Actual Body Weight applied to avoid overestimating clearance.";

            } else if (bmi >= 30) {

                recNote = "BMI ≥ 30 (Obesity): Actual Weight reported (MDCalc standard). Adjusted Weight available in comparison grid.";

            } else {

                recNote = "Standard weight range: Cockcroft-Gault uses Actual Body Weight (ABW - Standard).";

            }

            document.getElementById('weight-recommendation-note').textContent = recNote;

 

            const crcl = crcl_actual;

            const crclBsa = crcl * (1.73 / bsa_mosteller);

 

            // 6. CKD-EPI 2021 eGFR Calculation (Race-free equation)

            const kappa = isFemale ? 0.7 : 0.9;

            const alpha = isFemale ? -0.241 : -0.302;

            const sexFactorEpi = isFemale ? 1.012 : 1.0;

 

            const scrOverKappa = scr_mgdl / kappa;

            const minTerm = Math.pow(Math.min(scrOverKappa, 1), alpha);

            const maxTerm = Math.pow(Math.max(scrOverKappa, 1), -1.200);

            const ageTerm = Math.pow(0.9938, age);

 

            const egfr = 142 * minTerm * maxTerm * ageTerm * sexFactorEpi;

            const egfr_absolute = egfr * (bsa_mosteller / 1.73);

 

            // 7. Oncology Chemotherapy Calculations

            const cisplatin_dose_mosteller = 40 * bsa_mosteller;

            const cisplatin_dose_dubois = 40 * bsa_dubois;

 

            state.currentCisplatinDose = Math.round(cisplatin_dose_mosteller);

            state.currentBsa = bsa_mosteller;

 

            // Carboplatin Calvert AUC-2 Dose (FDA GFR Cap at 125 mL/min)

            const gfrUsedInCalvert = Math.min(125, crcl);

            const carboplatin_target_auc = 2;

            const carboplatin_dose = carboplatin_target_auc * (gfrUsedInCalvert + 25);

 

            // Update Primary Clearance UI

            document.getElementById('crcl-val').textContent = crcl.toFixed(1);

            document.getElementById('crcl-bsa-val').textContent = `${crclBsa.toFixed(1)} mL/min/1.73m²`;

            document.getElementById('crcl-weight-used').textContent = `Using: Actual Body Weight (${weight_kg.toFixed(1)} kg)`;

           

            document.getElementById('crcl-val-actual').textContent = crcl_actual.toFixed(1);

            document.getElementById('crcl-val-ibw').textContent = crcl_ibw.toFixed(1);

            document.getElementById('crcl-val-adj').textContent = crcl_adj.toFixed(1);

 

            document.getElementById('egfr-val').textContent = egfr.toFixed(1);

            document.getElementById('egfr-abs-val').textContent = `${egfr_absolute.toFixed(1)} mL/min`;

 

            // Update Cisplatin UI

            document.getElementById('cisplatin-dose-val').textContent = Math.round(cisplatin_dose_mosteller);

            document.getElementById('cisplatin-bsa-ref').textContent = `Based on Mosteller BSA (${bsa_mosteller.toFixed(2)} m²)`;

            document.getElementById('cisplatin-dubois-val').textContent = `${Math.round(cisplatin_dose_dubois)} mg`;

 

            // Update Carboplatin UI

            document.getElementById('carboplatin-dose-val').textContent = Math.round(carboplatin_dose);

            document.getElementById('calvert-gfr-note').textContent = `Effective CrCl used: ${gfrUsedInCalvert.toFixed(1)} mL/min`;

            document.getElementById('calvert-cap-status').textContent = crcl > 125 ? "FDA GFR Cap (125 mL/min) Active" : "FDA GFR Cap (< 125 mL/min)";

 

            // Update BSA Displays

            document.getElementById('bsa-mosteller').textContent = `${bsa_mosteller.toFixed(2)} m²`;

            document.getElementById('bsa-dubois').textContent = `${bsa_dubois.toFixed(2)} m²`;

 

            // Anthropometrics

            document.getElementById('metric-bmi').textContent = `${bmi.toFixed(1)} kg/m²`;

            document.getElementById('metric-bmi-cat').textContent = getBmiCategory(bmi);

            document.getElementById('metric-ibw').textContent = `${ibw.toFixed(1)} kg`;

            document.getElementById('metric-abw').textContent = `${abw.toFixed(1)} kg`;

 

            // Update CKD Stage Styling & Text

            updateCkdStageUI(egfr);

 

            // Update NCCN/ESMO Dose Modification Rules

            updateCisplatinModRules(crcl, bsa_mosteller);

 

            // Update Cumulative Tracker

            updateTracker();

 

            // Update Nutritional & Toxicity Risk Panel

            updateNutritionalRiskUI(bmi, weight_kg);

 

            // Math Expansion Formatting

            document.getElementById('math-bsa-expansion').textContent =

                `Mosteller = √(${height_cm.toFixed(1)} × ${weight_kg.toFixed(1)} / 3600) = ${bsa_mosteller.toFixed(2)} m² | Du Bois = 0.007184 × ${weight_kg.toFixed(1)}^0.425 × ${height_cm.toFixed(1)}^0.725 = ${bsa_dubois.toFixed(2)} m²`;

 

            document.getElementById('math-cg-expansion').textContent =

                `= [ (140 - ${age}) × ${weight_kg.toFixed(1)} kg ] / [ 72 × ${scr_mgdl.toFixed(2)} mg/dL ] ${isFemale ? '× 0.85' : ''} = ${crcl.toFixed(1)} mL/min`;

           

            document.getElementById('math-calvert-expansion').textContent =

                `= Target AUC (${carboplatin_target_auc}) × [ ${gfrUsedInCalvert.toFixed(1)} mL/min + 25 ] = ${carboplatin_dose.toFixed(1)} mg (${Math.round(carboplatin_dose)} mg)`;

 

            document.getElementById('math-cisplatin-expansion').textContent =

                `= 40 mg/m² × ${bsa_mosteller.toFixed(2)} m² (Mosteller) = ${cisplatin_dose_mosteller.toFixed(1)} mg (${Math.round(cisplatin_dose_mosteller)} mg)`;

 

            // Update EMR Note

            updateEmrNote({

                height_cm, weight_kg, bmi, bsa_mosteller, scr_mgdl, crcl, egfr, cisplatin_dose_mosteller, carboplatin_dose

            });

        }

        function getBmiCategory(bmi) {

            if (bmi < 18.5) return 'Underweight';

            if (bmi < 25.0) return 'Normal weight';

            if (bmi < 35.0) return 'Class I Obesity';

            if (bmi < 40.0) return 'Class II Obesity';

            return 'Class III Obesity';

        }
